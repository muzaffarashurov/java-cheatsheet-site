📌 Краткий конспект

JWT Flow — как работает аутентификация
Клиент → POST /api/auth/login (username/password) → AuthService → AuthenticationManager → JwtService → Token → Клиент хранит токен → Следующие запросы с Authorization: Bearer <token> → JwtAuthenticationFilter проверяет токен → доступ к защищённым ресурсам

📦 Шаг 1: Зависимости (pom.xml)

Добавь эти зависимости в <dependencies>. Maven автоматически скачает все библиотеки. Обрати внимание на Flyway и PostgreSQL.

pom.xml 
<!-- SPRING SECURITY — основа аутентификации и авторизации -->
<dependency>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-security</artifactId>
</dependency>

<!-- JJWT API — интерфейс для работы с JWT (генерация, парсинг) -->
<dependency>
    <groupId>io.jsonwebtoken</groupId>
    <artifactId>jjwt-api</artifactId>
    <version>0.12.6</version>
</dependency>

<!-- JJWT Implementation — реальная реализация (нужна при запуске) -->
<dependency>
    <groupId>io.jsonwebtoken</groupId>
    <artifactId>jjwt-impl</artifactId>
    <version>0.12.6</version>
    <scope>runtime</scope>
</dependency>

<!-- JJWT Jackson — сериализация payload в JSON -->
<dependency>
    <groupId>io.jsonwebtoken</groupId>
    <artifactId>jjwt-jackson</artifactId>
    <version>0.12.6</version>
    <scope>runtime</scope>
</dependency>

<!-- VALIDATION — аннотации @NotBlank, @Email для DTO -->
<dependency>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-validation</artifactId>
</dependency>

<!-- FLYWAY — версионирование базы данных (миграции SQL) -->
<dependency>
    <groupId>org.flywaydb</groupId>
    <artifactId>flyway-core</artifactId>
</dependency>

<!-- FLYWAY POSTGRESQL — диалект Flyway для PostgreSQL -->
<dependency>
    <groupId>org.flywaydb</groupId>
    <artifactId>flyway-database-postgresql</artifactId>
</dependency>

<!-- POSTGRESQL — драйвер БД -->
<dependency>
    <groupId>org.postgresql</groupId>
    <artifactId>postgresql</artifactId>
    <scope>runtime</scope>
</dependency>

⚙️ Шаг 2: Конфигурация (application.yml)

💡 Безопасность секрета
${JWT_SECRET:...} берёт значение из переменной окружения. В продакшене никогда не храни секрет в коде.
application.yml 
# =====================================================================
# application.yml — КОНФИГУРАЦИЯ приложения
# =====================================================================

spring:
  application:
    name: my_app
  
  # --- PostgreSQL Database ---
  datasource:
    url: jdbc:postgresql://localhost:5432/my_db?currentSchema=public
    username: ${DB_USER:postgres}  # Из переменной или дефолт
    password: ${DB_PASSWORD:postgres}
    driver-class-name: org.postgresql.Driver

  # --- JPA / Hibernate ---
  jpa:
    hibernate:
      ddl-auto: validate  # Только проверка! Схема управляется Flyway
    open-in-view: false   # Отключаем lazy-loading в слое View (антипаттерн)
    show-sql: true
    properties:
      hibernate:
        format_sql: true

  # --- Flyway Migrations ---
  flyway:
    enabled: true
    baseline-on-migrate: true  # Если БД не пустая, создать baseline
    locations: classpath:db/migration  # Где лежат .sql файлы

# --- Server ---
server:
  port: 8080

# --- JWT Settings ---
app:
  jwt:
    # Секрет минимум 256 бит для HS256
    secret: ${JWT_SECRET:mySuperSecretKeyForJwtTokenThatShouldBeVeryLongAndSecureAtLeast256Bits1234567890abcdef}
    # Время жизни токена: 86400000 мс = 24 часа
    expiration-ms: 86400000

User.java
package com.example.user.entity;

import com.example.common.entity.BaseEntity;  // Базовый класс с ID и датами
import jakarta.persistence.*;
import lombok.*;
import java.util.HashSet;
import java.util.Set;

@Entity
@Table(name = "users")
@Setter @Getter
@NoArgsConstructor @AllArgsConstructor
@Builder
public class User extends BaseEntity {  /* Наследуем id, createdAt, updatedAt */

    @Column(nullable = false, unique = true, length = 50)
    private String username;

    // Пароль хранится в захешированном виде (BCrypt)
    @Column(nullable = false, length = 255)
    private String password;

    @Column(nullable = false, unique = true, length = 150)
    private String email;

    @Column(name = "first_name", nullable = false, length = 100)
    private String firstName;

    @Column(name = "last_name", nullable = false, length = 100)
    private String lastName;

    @Builder.Default
    @Column(nullable = false)
    private boolean enabled = true;

    // ManyToMany связь с ролями через промежуточную таблицу user_roles
    @ManyToMany(fetch = FetchType.EAGER)  /* EAGER нужен для Security */
    @JoinTable(
        name = "user_roles",
        joinColumns = @JoinColumn(name = "user_id"),
        inverseJoinColumns = @JoinColumn(name = "role_id")
    )
    @Builder.Default
    private Set<Role> roles = new HashSet<>();
}

🔐 Шаг 7: JwtService (генерация и валидация токенов)

JwtService.java
package com.example.security.service;

import io.jsonwebtoken.*;
import io.jsonwebtoken.io.Decoders;
import io.jsonwebtoken.security.Keys;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.stereotype.Service;
import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;  /* Явная кодировка UTF-8 */
import java.util.Base64;
import java.util.Date;
import java.util.stream.Collectors;

@Service
@Slf4j
public class JwtService {
    
    // Читаем секрет из application.yml
    @Value("${app.jwt.secret}")
    private String jwtSecret;

    // Читаем время жизни из application.yml
    @Value("${app.jwt.expiration-ms}")
    private long jwtExpirationMs;

    // Создаём SecretKey для подписи (HMAC-SHA)
    private SecretKey key() {
        return Keys.hmacShaKeyFor(
            Decoders.BASE64.decode(
                Base64.getEncoder().encodeToString(jwtSecret.getBytes(StandardCharsets.UTF_8))
            )
        );
    }

    // Генерация токена после успешного логина
    public String generateToken(Authentication authentication) {
        String username = authentication.getName();
        
        // Собираем роли в строку через запятую
        String authorities = authentication.getAuthorities().stream()
            .map(GrantedAuthority::getAuthority)
            .collect(Collectors.joining(","));

        Date now = new Date();
        Date expiryDate = new Date(now.getTime() + jwtExpirationMs);

        return Jwts.builder()
            .subject(username)       // Поле sub: username
            .claim("roles", authorities) // Кастомное поле roles
            .issuedAt(now)          // Время создания
            .expiration(expiryDate) // Время истечения
            .signWith(key())        // Подпись секретным ключом
            .compact();              // Сериализация в строку
    }

    // Извлечение username из токена
    public String getUsernameFromToken(String token) {
        return parseToken(token).getSubject();
    }

    // Парсинг и верификация токена
    private Claims parseToken(String token) {
        return Jwts.parser()
            .verifyWith(key())      // Проверяем подпись
            .build()
            .parseSignedClaims(token) // Парсим payload
            .getPayload();
    }

    // Валидация: true если токен корректен и не просрочен
    public boolean validateToken(String token) {
        try {
            parseToken(token);
            return true;
        } catch (ExpiredJwtException e) {
            log.error("JWT expired: {}", e.getMessage());
        } catch (SecurityException e) {
            log.error("JWT invalid signature: {}", e.getMessage());
        } catch (Exception e) {
            log.error("JWT validation error: {}", e.getMessage());
        }
        return false;
    }
}

👤 Шаг 8: UserDetails (адаптер для Spring Security)

UserDetailsImpl.java
package com.example.security.service;

import com.example.user.entity.User;
import com.fasterxml.jackson.annotation.JsonIgnore;
import lombok.AllArgsConstructor;
import lombok.Data;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;
import java.util.Collection;
import java.util.List;
import java.util.stream.Collectors;

@Data
@AllArgsConstructor
public class UserDetailsImpl implements UserDetails {
    
    private static final long serialVersionUID = 1L;

    private Long id;
    private String username;
    private String email;

    // @JsonIgnore: не сериализовать пароль в JSON-ответах
    @JsonIgnore
    private String password;

    private Collection<? extends GrantedAuthority> authorities;
    private boolean enabled;

    // Фабричный метод: конвертация Entity → UserDetails
    public static UserDetailsImpl build(User user) {
        List<GrantedAuthority> authorities = user.getRoles().stream()
            .map(role -> new SimpleGrantedAuthority(role.getName()))
            .collect(Collectors.toList());

        return new UserDetailsImpl(
            user.getId(),
            user.getUsername(),
            user.getEmail(),
            user.getPassword(),
            authorities,
            user.isEnabled()
        );
    }

    @Override public Collection<? extends GrantedAuthority> getAuthorities() { return authorities; }
    @Override public String getPassword() { return password; }
    @Override public String getUsername() { return username; }
    @Override public boolean isAccountNonExpired() { return true; }
    @Override public boolean isAccountNonLocked() { return true; }
    @Override public boolean isCredentialsNonExpired() { return true; }
    @Override public boolean isEnabled() { return enabled; }
}
UserDetailsServiceImpl.java
package com.example.security.service;

import com.example.user.entity.User;
import com.example.user.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.core.userdetails.UsernameNotFoundException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
@Slf4j
public class UserDetailsServiceImpl implements UserDetailsService {
    
    private final UserRepository userRepository;

    // Вызывается Spring Security при логине для загрузки пользователя
    @Override
    @Transactional(readOnly = true)
    public UserDetails loadUserByUsername(String username) throws UsernameNotFoundException {
        User user = userRepository.findByUsername(username)
            .orElseThrow(() -> new UsernameNotFoundException(
                "User not found: " + username
            ));
        return UserDetailsImpl.build(user);
    }
}

⚙️ Шаг 9: AuthService (бизнес-логика)

AuthService.java
package com.example.security.service;

import com.example.security.dto.JwtResponse;
import com.example.security.dto.LoginRequest;
import com.example.security.dto.SignupRequest;
import com.example.user.entity.Role;
import com.example.user.entity.User;
import com.example.user.repository.RoleRepository;
import com.example.user.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Slf4j
public class AuthService {
    
    private final AuthenticationManager authenticationManager;
    private final UserRepository userRepository;
    private final RoleRepository roleRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtService jwtService;

    // АУТЕНТИФИКАЦИЯ (ЛОГИН)
    @Transactional
    public JwtResponse authenticate(LoginRequest request) {
        // 1. Проверяем логин/пароль через AuthenticationManager
        Authentication authentication = authenticationManager.authenticate(
            new UsernamePasswordAuthenticationToken(
                request.getUsername(), 
                request.getPassword()
            )
        );

        // 2. Сохраняем аутентификацию в контекст
        SecurityContextHolder.getContext().setAuthentication(authentication);

        // 3. Генерируем JWT
        String jwt = jwtService.generateToken(authentication);

        // 4. Формируем ответ
        UserDetailsImpl userDetails = (UserDetailsImpl) authentication.getPrincipal();
        List<String> roles = userDetails.getAuthorities().stream()
            .map(GrantedAuthority::getAuthority)
            .collect(Collectors.toList());

        return new JwtResponse(
            "Bearer", jwt, userDetails.getId(),
            userDetails.getUsername(), userDetails.getEmail(), roles
        );
    }

    // РЕГИСТРАЦИЯ
    @Transactional
    public void register(SignupRequest request) {
        // Проверка уникальности
        if (userRepository.existsByUsername(request.getUsername())) {
            throw new IllegalArgumentException("Username already taken");
        }
        if (userRepository.existsByEmail(request.getEmail())) {
            throw new IllegalArgumentException("Email already taken");
        }

        // Хешируем пароль перед сохранением
        User user = User.builder()
            .username(request.getUsername())
            .password(passwordEncoder.encode(request.getPassword()))
            .email(request.getEmail())
            .firstName(request.getFirstName())
            .lastName(request.getLastName())
            .enabled(true)
            .build();

        // Назначаем роли
        Set<Role> roles = new HashSet<>();
        if (request.getRoles() == null || request.getRoles().isEmpty()) {
            // Если роли не указаны → даём ROLE_USER
            Role userRole = roleRepository.findByName("ROLE_USER")
                .orElseThrow(() -> new RuntimeException("ROLE_USER not found"));
            roles.add(userRole);
        } else {
            // Иначе ищем запрошенные роли
            request.getRoles().forEach(roleStr -> {
                String roleName = roleStr.startsWith("ROLE_") ? roleStr : "ROLE_" + roleStr.toUpperCase();
                Role role = roleRepository.findByName(roleName)
                    .orElseThrow(() -> new RuntimeException("Role not found: " + roleName));
                roles.add(role);
            });
        }

        user.setRoles(roles);
        userRepository.save(user);
    }
}

🛡️ Шаг 10: JwtAuthenticationFilter

JwtAuthenticationFilter.java
package com.example.security.filter;

import com.example.security.service.JwtService;
import com.example.security.service.UserDetailsServiceImpl;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.web.authentication.WebAuthenticationDetailsSource;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;
import java.io.IOException;

@Component
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {
    
    private final JwtService jwtService;
    private final UserDetailsServiceImpl userDetailsService;

    @Override
    protected void doFilterInternal(HttpServletRequest request, 
                                    HttpServletResponse response, 
                                    FilterChain filterChain) throws ServletException, IOException {
        
        String authHeader = request.getHeader("Authorization");

        // Если нет заголовка или он не начинается с "Bearer " — пропускаем
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            filterChain.doFilter(request, response);
            return;
        }

        // Извлекаем токен (убираем префикс "Bearer ")
        String jwt = authHeader.substring(7);

        // Если токен невалиден — пропускаем
        if (!jwtService.validateToken(jwt)) {
            filterChain.doFilter(request, response);
            return;
        }

        // Извлекаем username из токена
        String username = jwtService.getUsernameFromToken(jwt);

        // Если пользователь найден и ещё не аутентифицирован в текущем запросе
        if (username != null && SecurityContextHolder.getContext().getAuthentication() == null) {
            UserDetails userDetails = userDetailsService.loadUserByUsername(username);
            
            // Создаём объект аутентификации
            UsernamePasswordAuthenticationToken authToken = 
                new UsernamePasswordAuthenticationToken(
                    userDetails, 
                    null,  // Credentials не нужны, токен уже проверен
                    userDetails.getAuthorities()
                );
            
            // Устанавливаем детали запроса (IP, SessionID и т.д.)
            authToken.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
            
            // Сохраняем аутентификацию в контекст безопасности
            SecurityContextHolder.getContext().setAuthentication(authToken);
        }

        // Продолжаем цепочку фильтров
        filterChain.doFilter(request, response);
    }
}

SecurityConfig.java
package com.example.security.config;

import com.example.security.filter.JwtAuthenticationFilter;
import com.example.security.service.UserDetailsServiceImpl;
import lombok.RequiredArgsConstructor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.authentication.AuthenticationProvider;
import org.springframework.security.authentication.dao.DaoAuthenticationProvider;
import org.springframework.security.config.annotation.authentication.configuration.AuthenticationConfiguration;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

@Configuration
@EnableWebSecurity
@EnableMethodSecurity(prePostEnabled = true, securedEnabled = true, jsr250Enabled = true)
@RequiredArgsConstructor
public class SecurityConfig {
    
    private final JwtAuthenticationFilter jwtAuthFilter;
    private final UserDetailsServiceImpl userDetailsService;

    // Настройка цепочки фильтров безопасности
    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            // 1. Отключаем CSRF (не нужен для JWT)
            .csrf(AbstractHttpConfigurer::disable)
            // 2. Отключаем форму логина и Basic Auth (используем JWT)
            .formLogin(AbstractHttpConfigurer::disable)
            .httpBasic(AbstractHttpConfigurer::disable)
            // 3. Stateless сессии (сервер не хранит состояние)
            .sessionManagement(session -> 
                session.sessionCreationPolicy(SessionCreationPolicy.STATELESS)
            )
            // 4. Провайдер аутентификации
            .authenticationProvider(authenticationProvider())
            // 5. Добавляем JWT-фильтр ПЕРЕД стандартным фильтром логина
            .addFilterBefore(jwtAuthFilter, UsernamePasswordAuthenticationFilter.class)
            // 6. Правила доступа к URL
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/api/v1/auth/**").permitAll()  // Логин/регистрация открыты
                .requestMatchers("/swagger-ui/**", "/v3/api-docs/**").permitAll() // Swagger открыт
                
                // Примеры ролей (раскомментируй при необходимости):
                // .requestMatchers(HttpMethod.GET, "/api/v1/cheatsheets/**").permitAll()
                // .requestMatchers("/api/v1/manager/**").hasAnyRole("MANAGER", "ADMIN")
                // .requestMatchers("/api/v1/admin/**").hasRole("ADMIN")
                
                // Всё остальное требует аутентификации
                .anyRequest().authenticated()
            );
        
        return http.build();
    }

    // Провайдер: связывает UserDetailsService и PasswordEncoder
    @Bean
    public AuthenticationProvider authenticationProvider() {
        DaoAuthenticationProvider authProvider = new DaoAuthenticationProvider();
        authProvider.setUserDetailsService(userDetailsService);
        authProvider.setPasswordEncoder(passwordEncoder());
        return authProvider;
    }

    // Менеджер аутентификации (нужен для AuthService)
    @Bean
    public AuthenticationManager authenticationManager(AuthenticationConfiguration config) throws Exception {
        return config.getAuthenticationManager();
    }

    // Кодировщик паролей BCrypt
    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }
}

✅ Best & ⚠️ Anti Practices

✅ Best Practices
  • Храни JWT_SECRET в переменных окружения, не в коде
  • Используй Flyway для версионирования схемы БД
  • Настраивай ddl-auto: validate (не update/create)
  • Используй @Transactional(readOnly = true) для SELECT
  • Хэшируй пароли через BCryptPasswordEncoder
  • Проверяй уникальность username/email до сохранения
  • Отключай CSRF для JWT-аутентификации
⚠️ Anti Patterns
  • ❌ Хранить пароли в открытом виде в БД
  • ❌ Использовать слабый секрет (<32 байт) для JWT
  • ❌ Возвращать UserDetails/Entity напрямую в JSON
  • ❌ Игнорировать валидацию (@Valid) на входящих DTO
  • ❌ Писать бизнес-логику в контроллере
  • ❌ Включать open-in-view в JPA (антипаттерн)
  • ❌ Отключать @PrePersist для createdAt/updatedAt

❓ Вопросы для собеседования

Как работает аутентификация с JWT в Spring Security?
How does JWT authentication work in Spring Security?
Medium ▼
✅ Ответ
1) Клиент отправляет username/password на /api/auth/login. 2) AuthenticationManager проверяет учётные данные через UserDetailsService + PasswordEncoder. 3) При успехе JwtService генерирует подписанный токен с username и ролями. 4) Клиент сохраняет токен и отправляет его в заголовке Authorization: Bearer <token>. 5) JwtAuthenticationFilter перехватывает каждый запрос, валидирует токен и устанавливает аутентификацию в SecurityContext. 6) Spring Security проверяет права доступа через @PreAuthorize или конфигурацию SecurityFilterChain.
Зачем нужен UserDetails и почему нельзя использовать User-Entity напрямую?
Why do we need UserDetails? Why not use User entity directly?
Easy ▼
✅ Ответ
UserDetails — это интерфейс Spring Security, который определяет контракт для аутентификации: методы getUsername(), getPassword(), getAuthorities(), isEnabled() и т.д. Наша User-entity содержит бизнес-поля (email, firstName, createdAt), которые не нужны для аутентификации. UserDetailsImpl выступает адаптером: преобразует Set<Role> в Collection<GrantedAuthority>, скрывает пароль через @JsonIgnore, предоставляет стандартный интерфейс для Security.
Почему мы отключаем CSRF в SecurityConfig для JWT?
Why disable CSRF in SecurityConfig for JWT?
Medium ▼
✅ Ответ
CSRF (Cross-Site Request Forgery) защищает от атак, когда злоумышленник заставляет браузер жертвы отправить запрос на доверенный сайт. Защита основана на токенах в cookie. Но при JWT-аутентификации мы используем stateless сессии и передаём токен в заголовке Authorization, а не в cookie. Браузер не отправляет заголовки автоматически при переходе по ссылке, поэтому атаки CSRF невозможны. Отключение CSRF безопасно для API, но обязательно включай его для веб-приложений с cookie-сессиями.
Как обеспечить безопасность JWT-секрета в продакшене?
How to secure JWT secret in production?
Hard ▼
✅ Ответ
1) Никогда не храни секрет в коде — используй переменные окружения: ${JWT_SECRET:...}. 2) В Kubernetes используй Secrets, в Docker — .env файлы (добавь в .gitignore). 3) Используй менеджеры секретов: HashiCorp Vault, AWS Secrets Manager. 4) Регулярно ротируй секреты (раз в 90 дней). 5) Используй минимум 256-битный ключ для HS256. 6) В CI/CD передавай секреты через защищённые переменные (GitHub Actions Secrets, GitLab CI Variables). 7) Логируй попытки использования невалидных токенов для мониторинга атак.
